Deployment guide & required sheet columns
  1. Create a Google Sheet and rename its first tab to PAYMENTS.
  2. In row 1, enter the 21 column headers exactly as listed in the script below (same order).
  3. Open Extensions → Apps Script, paste the script, and save.
  4. Deploy → New deployment → Web app — Execute as: Me, Who has access: Anyone. Deploy and copy the /exec URL.
  5. Paste the URL above (or into GOOGLE_SCRIPT_URL in index.html) and save.
const SHEET_NAME = "PAYMENTS";
const HEADERS = ["Receipt No","Date","Roll No","Student Name","Mobile No","Course","Duration","Payment Type","Payment Purpose","Month","Admission Fee","Registration Fee","Monthly Fee","Amount","Discount %","Discount Amount","Payable Amount","Amount Paid","Due Amount","Payment Mode","Remarks"];
const FIELDS = ["receiptNo","date","rollNo","studentName","mobile","course","duration","paymentType","paymentPurpose","month","admissionFee","registrationFee","monthlyFee","amount","discountPercent","discountAmount","payableAmount","amountPaid","dueAmount","paymentMode","remarks"];

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sh = ss.getSheetByName(SHEET_NAME);
    if (!sh) sh = ss.insertSheet(SHEET_NAME);
    if (sh.getLastRow() === 0) sh.appendRow(HEADERS);
    sh.appendRow(FIELDS.map(function (f) {
      return data[f] === undefined ? "" : data[f];
    }));
    return ContentService
      .createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: String(err) }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
